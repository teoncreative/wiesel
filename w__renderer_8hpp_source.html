46template <typename T>
47class Setting {
48 public:
49 Setting(T v) : value(v), change_hook(nullptr) {}
50
51 void SetHook(bool* ptr) { change_hook = ptr; }
52
53 T Get() { return value; }
54
55 Setting& operator=(const T& new_val) {
56 SetValue(new_val);
57 return *this;
58 }
59
60 operator T() const { return value; }
61
62 struct Proxy {
63 Setting* owner;
64 T old_value;
65
66 operator T*() { return &owner->value; }
67
68 ~Proxy() {
69 if (owner->change_hook && old_value != owner->value) {
70 *owner->change_hook = true;
71 }
72 }
73 };
74
75 Proxy operator&() { return Proxy{this, value}; }
76
77 private:
78 void SetValue(const T& new_val) {
79 if (value == new_val) {
80 return;
81 }
82
83 value = new_val;
84
85 if (change_hook) {
86 *change_hook = true;
87 }
88 }
89
90 T value;
91 bool* change_hook;
92};
