19// Interface for type-specific asset loaders.
20// Each asset type (Model, Texture, Audio, SpriteSheet) registers a loader.
21class IAssetLoader {
22 public:
23 virtual ~IAssetLoader() = default;
24
25 // Load the asset synchronously. Called from either the main thread
26 // (sync load) or a background thread (async load).
27 // Returns true on success.
28 virtual bool Load(AssetHandle handle) = 0;
29
30 // Unload the asset, freeing GPU/memory resources.
31 virtual void Unload(AssetHandle handle) = 0;
32};
33
34// Convenience: wraps a load function into an IAssetLoader.
35class FunctionAssetLoader : public IAssetLoader {
36 public:
37 using LoadFn = std::function<bool(AssetHandle)>;
38 using UnloadFn = std::function<void(AssetHandle)>;
39
40 FunctionAssetLoader(LoadFn load_fn, UnloadFn unload_fn = nullptr)
41 : load_fn_(std::move(load_fn)), unload_fn_(std::move(unload_fn)) {}
42
43 bool Load(AssetHandle handle) override { return load_fn_(handle); }
44
45 void Unload(AssetHandle handle) override {
46 if (unload_fn_) {
47 unload_fn_(handle);
48 }
49 }
50
51 private:
52 LoadFn load_fn_;
53 UnloadFn unload_fn_;
54};
