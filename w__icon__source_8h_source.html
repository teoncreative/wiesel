17// RGBA pixel data for a single icon, scaled to a specific size.
18// Caller is responsible for freeing pixels.
19struct IconBitmap {
20 uint8_t* pixels = nullptr; // RGBA, row-major
21 int width = 0;
22 int height = 0;
23};
24
25// Interface for icon glyph sources. Implementations provide RGBA bitmaps
26// for icon codepoints at any requested size.
27class IIconSource {
28 public:
29 virtual ~IIconSource() = default;
30
31 // Returns true if this source can provide a glyph for the given codepoint.
32 virtual bool HasGlyph(uint32_t codepoint) const = 0;
33
34 // Rasterize the icon at the given pixel size. Returns an IconBitmap with
35 // newly allocated pixels (caller frees with free()). Returns empty on failure.
36 virtual IconBitmap RasterizeGlyph(uint32_t codepoint, int size) const = 0;
37};
38
39// Icon source that loads PNG files from VFS and scales them on demand.
40class PngIconSource : public IIconSource {
41 public:
42 // Register a PNG icon for a codepoint. Loads the PNG from VFS immediately.
43 void AddIcon(uint32_t codepoint, const std::string& vfs_path);
44
45 bool HasGlyph(uint32_t codepoint) const override;
46 IconBitmap RasterizeGlyph(uint32_t codepoint, int size) const override;
47
48 private:
49 struct SourceImage {
50 std::vector<uint8_t> pixels; // RGBA
51 int width = 0;
52 int height = 0;
53 };
54
55 std::unordered_map<uint32_t, SourceImage> icons_;
56};
