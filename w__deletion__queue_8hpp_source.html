18class DeletionQueue {
19 public:
20 using DeleteFn = std::function<void()>;
21
22 void Push(DeleteFn&& fn, uint32_t frames_to_wait = 2) {
23 entries_.push_back({std::move(fn), frames_to_wait});
24 }
25
26 // Call once per frame after presenting. Ticks down all counters and
27 // executes deletions whose wait has expired.
28 // Safe against re-entrant Push() calls from within fn().
29 void Flush() {
30 // Swap out so new entries pushed during execution don't invalidate iteration
31 std::vector<Entry> current;
32 std::swap(current, entries_);
33
34 for (auto& entry : current) {
35 if (entry.frames_remaining == 0) {
36 entry.fn();
37 } else {
38 --entry.frames_remaining;
39 entries_.push_back(std::move(entry));
40 }
41 }
42 }
43
44 // Force-delete everything immediately (e.g. on shutdown).
45 // Keeps flushing until empty since deletions can cascade.
46 void FlushAll() {
47 while (!entries_.empty()) {
48 std::vector<Entry> current;
49 std::swap(current, entries_);
50 for (auto& entry : current) {
51 entry.fn();
52 }
53 }
54 }
55
56 private:
57 struct Entry {
58 DeleteFn fn;
59 uint32_t frames_remaining;
60 };
61
62 std::vector<Entry> entries_;
63};
