103template <typename T>
104void AssetManager::Store(AssetHandle handle, std::shared_ptr<T> resource) {
105 auto it = registry_.find(handle);
106 if (it == registry_.end()) {
107 LOG_ERROR("AssetManager::Store called with unregistered handle {}",
108 handle.ToString());
109 return;
110 }
111 it->second->resource = std::static_pointer_cast<void>(resource);
112}
113
114template <typename T>
115std::shared_ptr<T> AssetManager::Get(AssetHandle handle) const {
116 auto it = registry_.find(handle);
117 if (it == registry_.end() || !it->second->resource) {
118 return nullptr;
119 }
120 return std::static_pointer_cast<T>(it->second->resource);
121}

123template <typename T>
124std::shared_ptr<T> AssetManager::GetOrLoad(AssetHandle handle) const {
125 auto it = registry_.find(handle);
126 if (it == registry_.end()) {
127 return nullptr;
128 }
129 if (!it->second->resource) {
130 Engine::LoadModelAsync(handle);
131 return nullptr;
132 }
133 return std::static_pointer_cast<T>(it->second->resource);
134}

136template <typename T>
137AssetHandle AssetManager::RegisterAndStore(const std::string& name,
138 AssetType type,
139 const std::string& virtual_source_path,
140 std::shared_ptr<T> resource) {
141 AssetHandle handle = Register(name, type, virtual_source_path);
142 if (handle.IsValid()) {
143 Store<T>(handle, std::move(resource));
144 SetLoadState(handle, AssetLoadState::Unloaded, AssetLoadState::Loaded);
145 }
146 return handle;
147}
