163template <typename T>
164void AssetManager::Store(AssetHandle handle, std::shared_ptr<T> resource) {
165 if (!handle.IsValid()) {
166 DCON_LOG_ERROR("AssetManager::Store called with invalid handle");
167 return;
168 }
169 auto it = registry_.find(handle);
170 if (it == registry_.end()) {
171 DCON_LOG_ERROR("AssetManager::Store called with unregistered handle {}",
172 handle.ToString());
173 return;
174 }
175 it->second->resource = std::static_pointer_cast<void>(resource);
176}
177
178template <typename T>
179std::shared_ptr<T> AssetManager::Get(AssetHandle handle) const {
180 if (!handle.IsValid()) {
181 return nullptr;
182 }
183 auto it = registry_.find(handle);
184 if (it == registry_.end() || !it->second->resource) {
185 return nullptr;
186 }
187 return std::static_pointer_cast<T>(it->second->resource);
188}
189
190template <typename T>
191std::shared_ptr<T> AssetManager::GetOrLoad(AssetHandle handle) const {
192 if (!handle.IsValid()) {
193 return nullptr;
194 }
195 auto it = registry_.find(handle);
196 if (it == registry_.end()) {
197 return nullptr;
198 }
199 if (!it->second->resource) {
200 // Trigger async load via the registered loader
201 const_cast<AssetManager*>(this)->LoadAsync(handle);
202 return nullptr;
203 }
204 return std::static_pointer_cast<T>(it->second->resource);
205}

207template <typename T>
208AssetHandle AssetManager::RegisterAndStore(
209 const std::string& name, AssetType type,
210 const std::string& virtual_source_path, std::shared_ptr<T> resource) {
211 AssetHandle handle = Register(name, type, virtual_source_path);
212 if (handle.IsValid()) {
213 Store<T>(handle, std::move(resource));
214 SetLoadState(handle, AssetLoadState::Unloaded, AssetLoadState::Loaded);
215 }
216 return handle;
217}
