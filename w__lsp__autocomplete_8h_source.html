18class LspAutocompleteProvider : public TextEditor::IAutocompleteProvider {

22 void RequestCompletions(const std::string& filePath, int line, int column,
23 const std::string& prefix) override {

26 }
27
28 void OnTextChanged(const std::string& filePath,
29 const std::string& fullText) override {
30 std::string uri = LspClient::PathToUri(filePath);
31 client_.DidChange(uri, fullText);
32 }
33
34 bool HasResults() override { return client_.HasCompletions(); }
35
36 std::vector<TextEditor::CompletionItem> TakeResults() override {
37 auto lsp_items = client_.TakeCompletions();
38 std::vector<TextEditor::CompletionItem> items;
39 items.reserve(lsp_items.size());
40 for (auto& li : lsp_items) {
41 items.push_back({li.label, li.detail, li.insert_text,
42 static_cast<TextEditor::CompletionKind>(li.kind)});
43 }
44 return items;
45 }
46
47 private:
48 LspClient& client_;
49};
