39 // Transform local-space AABB to world-space AABB
40 AABB Transformed(const glm::mat4& m) const {
41 glm::vec3 center = Center();
42 glm::vec3 extents = Extents();
43 glm::vec3 new_center = glm::vec3(m * glm::vec4(center, 1.0f));
44 glm::vec3 new_extents(
45 std::abs(m[0][0]) * extents.x + std::abs(m[1][0]) * extents.y +
46 std::abs(m[2][0]) * extents.z,
47 std::abs(m[0][1]) * extents.x + std::abs(m[1][1]) * extents.y +
48 std::abs(m[2][1]) * extents.z,
49 std::abs(m[0][2]) * extents.x + std::abs(m[1][2]) * extents.y +
50 std::abs(m[2][2]) * extents.z);
51 return {new_center - new_extents, new_center + new_extents};
52 }
