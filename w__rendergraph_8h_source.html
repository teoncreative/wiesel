26// Opaque handle to a resource in the render graph
27struct RGResource {
28 uint32_t index = UINT32_MAX;
29
30 bool IsValid() const { return index != UINT32_MAX; }
31
32 bool operator==(const RGResource& other) const {
33 return index == other.index;
34 }
35
36 bool operator!=(const RGResource& other) const {
37 return index != other.index;
38 }
39};

41// How a pass accesses a resource
42enum class RGAccess {
43 ColorAttachmentWrite, // VK_IMAGE_LAYOUT_COLOR_ATTACHMENT_OPTIMAL
44 DepthStencilWrite, // VK_IMAGE_LAYOUT_DEPTH_STENCIL_ATTACHMENT_OPTIMAL
45 DepthStencilReadOnly, // VK_IMAGE_LAYOUT_DEPTH_STENCIL_READ_ONLY_OPTIMAL
46 ShaderRead, // VK_IMAGE_LAYOUT_SHADER_READ_ONLY_OPTIMAL
47 StorageImageRead, // VK_IMAGE_LAYOUT_GENERAL (imageLoad in shader)

49 Present // VK_IMAGE_LAYOUT_PRESENT_SRC_KHR
