19// Runtime data for a .wsprite asset - a sub-region of a texture.
20struct SpriteAssetData {
21 AssetHandle texture_handle;
22 glm::vec4 rect = {0, 0, 0, 0}; // x, y, width, height in pixels
23 glm::vec2 pivot = {0.5f, 0.5f};
24
25 // Compute UV rect from pixel rect and texture dimensions.
26 glm::vec4 GetUVRect(float tex_width, float tex_height) const {
27 if (tex_width <= 0 || tex_height <= 0) {
28 return {0, 0, 1, 1};
29 }
30 return {rect.x / tex_width, rect.y / tex_height, rect.z / tex_width,
31 rect.w / tex_height};
32 }
33};

46// Runtime data for a .wspritecontroller asset - state machine.

58 const State* FindState(const std::string& name) const {
59 for (const auto& s : states) {
