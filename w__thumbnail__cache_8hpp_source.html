22struct ThumbnailEntry {
23 VkDescriptorSet texture_id = nullptr;
24 bool attempted = false;
25 uint32_t width = 0;
26 uint32_t height = 0;
27 ImVec2 uv0 = {0.0f, 0.0f};
28 ImVec2 uv1 = {1.0f, 1.0f};
29
30 ImVec2 FitSize(float max_size) const {
31 float w = static_cast<float>(width) * (uv1.x - uv0.x);
32 float h = static_cast<float>(height) * (uv1.y - uv0.y);
33 if (w <= 0 || h <= 0) {
34 return ImVec2(max_size, max_size);
35 }
36 float aspect = w / h;
37 if (aspect >= 1.0f) {
38 return ImVec2(max_size, max_size / aspect);
39 }
40 return ImVec2(max_size * aspect, max_size);
41 }
42
43 // Visible dimensions (pixel size of the sub-region)
44 uint32_t VisibleWidth() const {
45 return static_cast<uint32_t>(width * (uv1.x - uv0.x));
46 }
47
48 uint32_t VisibleHeight() const {
49 return static_cast<uint32_t>(height * (uv1.y - uv0.y));
50 }
51};
