48struct MaterialProperty {
49 MaterialProperty() = default;
50 MaterialProperty(std::string name, std::string display_name,
51 MaterialPropertyType type, MaterialPropertyValue default_value,
52 float min_value = 0.0f, float max_value = 1.0f)

54 type(type), default_value(std::move(default_value)),
55 min_value(min_value), max_value(max_value) {}

57 std::string name;
58 std::string display_name; // for editor UI
59 MaterialPropertyType type = MaterialPropertyType::Float;
60 MaterialPropertyValue default_value = 0.0f;
61 float min_value = 0.0f; // for float/int slider range
62 float max_value = 1.0f;
63};
