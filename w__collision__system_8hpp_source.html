21// Ordered pair of entities (lower handle first) to avoid duplicates
22struct CollisionPair {
23 entt::entity a;
24 entt::entity b;
25
26 CollisionPair(entt::entity x, entt::entity y) {
27 if (x < y) {
28 a = x;
29 b = y;
30 } else {
31 a = y;
32 b = x;
33 }
34 }
35
36 bool operator<(const CollisionPair& other) const {
37 if (a != other.a) {
38 return a < other.a;
39 }
40 return b < other.b;
41 }
42
43 bool operator==(const CollisionPair& other) const {
44 return a == other.a && b == other.b;
45 }
46};
