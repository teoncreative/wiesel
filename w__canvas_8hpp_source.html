113
114 // Texture per state (normal is required, others fall back to normal)

117 AssetHandle pressed_texture;

119 AssetHandle disabled_texture;

121 // Child offset applied per state (pixels, affects children positioning)

123 glm::vec2 pressed_offset = {0.0f, 0.0f};
124 glm::vec2 selected_offset = {0.0f, 0.0f};

126 // Runtime state (not serialized)
127 ButtonState state_ = ButtonState::Normal;
