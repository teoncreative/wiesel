60// Template implementation
61template <typename T>
62AssetHandle AssetSerializerRegistry::Create(const std::string& name,
63 AssetType type,
64 const std::string& vfs_path,
65 std::shared_ptr<T> data) {
66 auto& mgr = Engine::asset_manager();
67 AssetHandle handle = mgr.Register(name, type, vfs_path);
68 if (!handle.IsValid()) {
69 return {};
70 }
71 mgr.Store<T>(handle, data);
72 if (!Save(handle, vfs_path)) {
73 return {};
74 }
75 return handle;
76}
