87// Abstract base class for a render feature.
88// Each feature encapsulates a reusable rendering effect (shadow mapping,
89// SSAO, bloom, etc.) and owns its own GPU pipelines and render passes.
90class RenderFeature {
91 public:
92 virtual ~RenderFeature() = default;
93 virtual const std::string& GetName() const = 0;
94
95 // Create GPU resources (textures, framebuffers, descriptors) and store
96 // them in ctx.resources. Called when camera resources are dirty.
97 virtual void SetupResources(RenderContext& ctx) = 0;
98
99 // Add passes to the render graph. Import resources from the pool,
100 // wire reads/writes, and register outputs in the registry.
101 virtual void AddPasses(RenderGraph& graph,
102 RenderResourceRegistry& registry,
103 RenderContext& ctx) = 0;
104
105 // Whether this feature should run this frame.
106 virtual bool IsEnabled(const RenderContext& ctx) const { return true; }
107};

109// Ordered list of features = a complete render pipeline.
110// Applications create a RenderPipeline, add features to it, and assign
111// it to the scene or to individual cameras.
112class RenderPipeline {
113 public:
114 explicit RenderPipeline(std::shared_ptr<Renderer> renderer);
115
116 template <typename T, typename... Args>
117 T& AddFeature(Args&&... args) {
118 auto feature = std::make_shared<T>(std::forward<Args>(args)...);
119 T& ref = *feature;
120 features_.push_back(std::move(feature));
121 return ref;
122 }
123
124 void RemoveFeature(const std::string& name);
125
126 // Setup all feature resources for a camera.
127 void SetupResources(RenderContext& ctx);
128
129 // Build the render graph for a camera (calls each feature's AddPasses).
130 void BuildRenderGraph(RenderGraph& graph, RenderContext& ctx);
131
132 // Get the final output from the "PipelineOutput" convention.
133 std::shared_ptr<DescriptorSet> GetFinalOutputDescriptor(CameraResourcePool& pool) const;
134 std::shared_ptr<AttachmentTexture> GetFinalOutputImage(CameraResourcePool& pool) const;
135
136 const std::vector<std::shared_ptr<RenderFeature>>& GetFeatures() const {
137 return features_;
138 }
139
140 std::shared_ptr<Renderer> GetRenderer() const { return renderer_; }
141
142 private:
143 std::shared_ptr<Renderer> renderer_;
144 std::vector<std::shared_ptr<RenderFeature>> features_;
145};
