272inline const char* ToString(SamplingMode sampling_mode) {
273 switch (sampling_mode) {
274 case SamplingMode::DISABLED: return "Disabled";
275 case SamplingMode::X2: return "2x";
276 case SamplingMode::X4: return "4x";
277 case SamplingMode::X8: return "8x";
278 case SamplingMode::X16: return "16x";
279 case SamplingMode::X32: return "32x";
280 case SamplingMode::X64: return "64x";
281 default: return "Unknown";
282 }
283}
