61 return std::find(tags.begin(), tags.end(), tag) != tags.end();

63

66 tags.push_back(tag);
67 }
68 }
69
70 void RemoveTag(const std::string& tag) {

146
147 // Direction conversion between local and world space
148 glm::vec3 LocalToWorldDirection(const glm::vec3& dir) const {
149 return glm::normalize(glm::mat3(transform_matrix_) * dir);
150 }
151
152 glm::vec3 WorldToLocalDirection(const glm::vec3& dir) const {
153 return glm::normalize(glm::inverse(glm::mat3(transform_matrix_)) * dir);
154 }
155
156 // Point conversion between local and world space
157 glm::vec3 LocalToWorldPoint(const glm::vec3& point) const {
158 return glm::vec3(transform_matrix_ * glm::vec4(point, 1.0f));
159 }
160
161 glm::vec3 WorldToLocalPoint(const glm::vec3& point) const {
162 return glm::vec3(glm::inverse(transform_matrix_) * glm::vec4(point, 1.0f));
163 }

165 enum class Space { Local, World };
166
167 // Relative modifiers
168 void Move(const glm::vec3& delta) {
169 position_ += delta;
170 is_changed_ = true;
171 }
172
173 void Move(float dx, float dy, float dz) { Move({dx, dy, dz}); }
174
175 // Translate with space selection
176 void Translate(const glm::vec3& delta, Space space = Space::Local) {
177 if (space == Space::World) {
178 // Convert world-space delta to local space
179 glm::mat3 world_rot = glm::mat3(transform_matrix_);
180 glm::mat3 inv_rot = glm::inverse(world_rot);
181 position_ += inv_rot * delta;
182 } else {
183 position_ += delta;
184 }
185 is_changed_ = true;
186 }
187
188 void Rotate(const glm::vec3& delta) {
189 rotation_ += delta;
190 is_changed_ = true;
191 }
192
193 void Rotate(float dx, float dy, float dz) { Rotate({dx, dy, dz}); }
194
195 void Resize(const glm::vec3& delta) {
196 scale_ += delta;
197 is_changed_ = true;
198 }
199
200 void Resize(float dx, float dy, float dz) { Resize({dx, dy, dz}); }
201
202 // Internal: used by scene to update cached matrices and reset flag
203 void SetTransformMatrix(const glm::mat4& mat) {
204 transform_matrix_ = mat;
205 normal_matrix_ = glm::mat3(mat);
206 }
207
208 void ClearChanged() { is_changed_ = false; }
209
210 void MarkChanged() { is_changed_ = true; }
211
212 // Direct mutable access (for serialization, physics sync, editor inspector)
213 glm::vec3& PositionMut() { return position_; }

215 glm::vec3& RotationMut() { return rotation_; }
216
217 glm::vec3& ScaleMut() { return scale_; }
218
219 glm::vec3& PivotMut() { return pivot_; }
220
221 private:
222 glm::vec3 position_ = {0.0f, 0.0f, 0.0f};
223 glm::vec3 rotation_ = {0.0f, 0.0f, 0.0f}; // in degrees
224 glm::vec3 scale_ = {1.0f, 1.0f, 1.0f};
225 glm::vec3 pivot_ = {0.0f, 0.0f, 0.0f};
226 bool is_changed_ = true;
227 glm::mat4 transform_matrix_ = {};
228 glm::mat3 normal_matrix_ = {};
229};
