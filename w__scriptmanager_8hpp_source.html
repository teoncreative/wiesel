42class FieldData {
43 public:
44 FieldData(MonoClassField* field, const std::string& fieldName,
45 uint32_t fieldFlags) : field_(field),
46 field_name_(fieldName),
47 field_flags_(fieldFlags) {
48 std::string typeName = mono_type_get_name(mono_field_get_type(field_));
49 if (typeName == "System.Boolean") {
50 field_type_ = FieldType::Boolean;
51 } else if (typeName == "System.Single") {
52 field_type_ = FieldType::Float;
53 } else if (typeName == "System.Double") {
54 field_type_ = FieldType::Double;
55 } else if (typeName == "System.Int32") {
56 field_type_ = FieldType::Integer;
57 } else if (typeName == "System.Int64") {
58 field_type_ = FieldType::Long;
59 } else if (typeName == "System.UInt32") {
60 field_type_ = FieldType::UnsignedInteger;
61 } else if (typeName == "System.UInt64") {
62 field_type_ = FieldType::UnsignedLong;
63 } else if (typeName == "System.String") {
64 field_type_ = FieldType::String;
65 } else if (typeName == "WieselEngine.Entity") {
66 field_type_ = FieldType::Entity;
67 } else if (typeName == "WieselEngine.Prefab") {
68 field_type_ = FieldType::Prefab;
69 } else if (typeName == "WieselEngine.AudioClip") {
70 field_type_ = FieldType::AudioClip;
71 } else {
72 field_type_ = FieldType::Object;
73 LOG_WARN("Unknown script field type '{}' for field '{}'", typeName, fieldName);
74 }
75 formatted_name_ = FormatVariableName(field_name_);
76 }
77
78 template <typename T>
79 void Set(MonoObject* instance, T value) {
80 mono_field_set_value(instance, field_, value);
81 }
82
83 template <typename T>
84 T Get(MonoObject* instance) const {
85 T value;
86 mono_field_get_value(instance, field_, &value);
87 return value;
88 }
89
90 MonoClassField* field() const { return field_; }
91 const std::string& field_name() const { return field_name_; }
92 uint32_t field_flags() const { return field_flags_; }
93 FieldType field_type() const { return field_type_; }
94 const std::string& formatted_name() const { return formatted_name_; }
95
96 private:
97 MonoClassField* field_;
98 std::string field_name_;
99 std::string formatted_name_;
100 uint32_t field_flags_;
101 FieldType field_type_;
102};
