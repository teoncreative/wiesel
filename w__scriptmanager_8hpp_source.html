43class FieldData {
44 public:
45 FieldData(MonoClassField* field, const std::string& fieldName,
46 uint32_t fieldFlags)
47 : field_(field), field_name_(fieldName), field_flags_(fieldFlags) {
48 std::string typeName = mono_type_get_name(mono_field_get_type(field_));
49 if (typeName == "System.Boolean") {
50 field_type_ = FieldType::Boolean;
51 } else if (typeName == "System.Single") {
52 field_type_ = FieldType::Float;
53 } else if (typeName == "System.Double") {
54 field_type_ = FieldType::Double;
55 } else if (typeName == "System.Int32") {
56 field_type_ = FieldType::Integer;
57 } else if (typeName == "System.Int64") {
58 field_type_ = FieldType::Long;
59 } else if (typeName == "System.UInt32") {
60 field_type_ = FieldType::UnsignedInteger;
61 } else if (typeName == "System.UInt64") {
62 field_type_ = FieldType::UnsignedLong;
63 } else if (typeName == "System.String") {
64 field_type_ = FieldType::String;
65 } else if (typeName == "WieselEngine.Entity") {
66 field_type_ = FieldType::Entity;
67 } else if (typeName == "WieselEngine.Prefab") {
68 field_type_ = FieldType::Prefab;
69 } else if (typeName == "WieselEngine.AudioClip") {
70 field_type_ = FieldType::AudioClip;
71 } else {
72 field_type_ = FieldType::Object;
73 LOG_WARN("Unknown script field type '{}' for field '{}'", typeName,
74 fieldName);
75 }
76 formatted_name_ = FormatVariableName(field_name_);
77 }
78
79 template <typename T>
80 void Set(MonoObject* instance, T value) {
81 mono_field_set_value(instance, field_, value);
82 }
83
84 template <typename T>
85 T Get(MonoObject* instance) const {
86 T value;
87 mono_field_get_value(instance, field_, &value);
88 return value;
89 }
90
91 MonoClassField* field() const { return field_; }
92
93 const std::string& field_name() const { return field_name_; }
94
95 uint32_t field_flags() const { return field_flags_; }
96
97 FieldType field_type() const { return field_type_; }
98
99 const std::string& formatted_name() const { return formatted_name_; }
100
101 private:
102 MonoClassField* field_;
103 std::string field_name_;
104 std::string formatted_name_;
105 uint32_t field_flags_;
106 FieldType field_type_;
107};
