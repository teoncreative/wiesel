123 // Compute mesh_node_transforms from node hierarchy (call after ProcessNode)
124 void ComputeMeshNodeTransforms() {
125 mesh_node_transforms.resize(meshes.size(), glm::mat4(1.0f));
126 for (size_t i = 0; i < meshes.size(); i++) {
127 int32_t ni = meshes[i]->node_index;
128 if (ni >= 0) {
129 mesh_node_transforms[i] = node_hierarchy.GetWorldTransform(ni);
130 }
131 }
132 }
