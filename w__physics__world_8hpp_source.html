33// Ordered contact pair for Enter/Stay/Exit tracking
34struct ContactPair {
35 entt::entity a;
36 entt::entity b;
37
38 ContactPair(entt::entity x, entt::entity y) {
39 if (x < y) {
40 a = x;
41 b = y;
42 } else {
43 a = y;
44 b = x;
45 }
46 }
47
48 bool operator<(const ContactPair& o) const {
49 if (a != o.a) {
50 return a < o.a;
51 }
52 return b < o.b;
53 }
54
55 bool operator==(const ContactPair& o) const { return a == o.a && b == o.b; }
56};
