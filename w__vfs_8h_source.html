87// Template implementation
88template <typename T>
89T VfsFile::Read() {
90 if (position_ + sizeof(T) > data_.size()) {
91 throw std::runtime_error("VfsFile::Read past end of file: " + path_);
92 }
93 T value;
94 std::memcpy(&value, data_.data() + position_, sizeof(T));
95 position_ += sizeof(T);
96 return value;
97}
