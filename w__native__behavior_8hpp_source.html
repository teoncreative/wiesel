51// Registry for native behaviors
52class NativeBehaviorRegistry {
53 public:
54 NativeBehaviorRegistry() = default;
55
56 template <typename T>
57 void Register(const std::string& name) {
58 factories_[name] = [name](Entity entity) -> NativeBehavior* {
59 return new T(entity, name);
60 };
61 names_.push_back(name);
62 }
63
64 bool Has(const std::string& name) const {
65 return factories_.contains(name);
66 }
67
68 NativeBehavior* Create(const std::string& name, Entity entity) const {
69 auto it = factories_.find(name);
70 if (it == factories_.end()) return nullptr;
71 return it->second(entity);
72 }
73
74 const std::vector<std::string>& GetNames() const { return names_; }
75
76 private:
77 std::unordered_map<std::string, NativeBehaviorFactory> factories_;
78 std::vector<std::string> names_;
79};
