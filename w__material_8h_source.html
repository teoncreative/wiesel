27// --- Material Property System ---

29enum class MaterialPropertyType { Float, Vec2, Vec3, Vec4, Texture, Bool, Int };

31using MaterialPropertyValue =
32 std::variant<float, glm::vec2, glm::vec3, glm::vec4,
33 std::shared_ptr<Texture>, bool, int32_t>;

35struct MaterialProperty {
36 MaterialProperty() = default;
37
38 MaterialProperty(std::string name, std::string display_name,
39 MaterialPropertyType type,
40 MaterialPropertyValue default_value, float min_value = 0.0f,
41 float max_value = 1.0f)
42 : name(std::move(name)),
43 display_name(std::move(display_name)),
44 type(type),
45 default_value(std::move(default_value)),
46 min_value(min_value),
47 max_value(max_value) {}
48
49 std::string name;
50 std::string display_name; // for editor UI
51 MaterialPropertyType type = MaterialPropertyType::Float;
52 MaterialPropertyValue default_value = 0.0f;
53 float min_value = 0.0f; // for float/int slider range
54 float max_value = 1.0f;
55};

96// --- Texture Slot ---
97// Holds an AssetHandle + lazily-resolved cached pointer.
98// Cache is invalidated when the asset's version changes (e.g. on reload).
99
100class AssetManager;
101
102struct TextureSlot {
103 AssetHandle handle;
104 mutable std::shared_ptr<Texture> cached;
105 mutable uint32_t resolved_version = 0;
106
107 TextureSlot() = default;
108
109 explicit TextureSlot(AssetHandle h) : handle(h) {}
110
111 // Resolve the handle to a Texture pointer.
112 // Returns true if the texture changed since last resolve (triggers descriptor rebuild).
113 bool Resolve(std::shared_ptr<Texture>& out) const;
114
115 // Check if the slot has a valid handle assigned.
116 bool HasTexture() const { return handle.IsValid(); }
117
118 // Clear the slot.
119 void Clear() {
120 handle = {};
121 cached.reset();
122 resolved_version = 0;
123 }
124};
