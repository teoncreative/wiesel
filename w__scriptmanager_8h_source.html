40class FieldData {
41 public:
42 FieldData(MonoClassField* field, const std::string& fieldName,
43 uint32_t fieldFlags)
44 : field_(field), field_name_(fieldName), field_flags_(fieldFlags) {
45 std::string typeName = mono_type_get_name(mono_field_get_type(field_));
46 if (typeName == "System.Boolean") {
47 field_type_ = FieldType::Boolean;
48 } else if (typeName == "System.Single") {
49 field_type_ = FieldType::Float;
50 } else if (typeName == "System.Double") {
51 field_type_ = FieldType::Double;
52 } else if (typeName == "System.Int32") {
53 field_type_ = FieldType::Integer;
54 } else if (typeName == "System.Int64") {
55 field_type_ = FieldType::Long;
56 } else if (typeName == "System.UInt32") {
57 field_type_ = FieldType::UnsignedInteger;
58 } else if (typeName == "System.UInt64") {
59 field_type_ = FieldType::UnsignedLong;
60 } else if (typeName == "System.String") {
61 field_type_ = FieldType::String;
62 } else if (typeName == "WieselEngine.Entity") {
63 field_type_ = FieldType::Entity;
64 } else if (typeName == "WieselEngine.Prefab") {
65 field_type_ = FieldType::Prefab;
66 } else if (typeName == "WieselEngine.AudioClip") {
67 field_type_ = FieldType::AudioClip;
68 } else {
69 field_type_ = FieldType::Object;
70 LOG_WARN("Unknown script field type '{}' for field '{}'", typeName,
71 fieldName);
72 }
73 formatted_name_ = FormatVariableName(field_name_);
74 }
75
76 template <typename T>
77 void Set(MonoObject* instance, T value) {
78 mono_field_set_value(instance, field_, value);
79 }
80
81 template <typename T>
82 T Get(MonoObject* instance) const {
83 T value;
84 mono_field_get_value(instance, field_, &value);
85 return value;
86 }
87
88 MonoClassField* field() const { return field_; }
89
90 const std::string& field_name() const { return field_name_; }
91
92 uint32_t field_flags() const { return field_flags_; }
93
94 FieldType field_type() const { return field_type_; }
95
96 const std::string& formatted_name() const { return formatted_name_; }
97
98 private:
99 MonoClassField* field_;
100 std::string field_name_;
101 std::string formatted_name_;
102 uint32_t field_flags_;
103 FieldType field_type_;
104};
