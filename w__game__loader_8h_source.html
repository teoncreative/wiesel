24// Runtime game loader - read-only asset scanning and configuration.
25// Used by standalone games. Never generates handles or writes files.
26class GameLoader {
27 public:
28 // Mount an assets directory to VFS /app.
29 static bool MountAssets(const std::filesystem::path& assets_dir);
30
31 // Scan the mounted /app directory for assets (read-only).
32 // Reads handles from .meta files and JSON asset files.
33 // Skips assets without pre-existing handles.
34 static void ScanAssets();
35
36 // Scan a single VFS prefix for assets using a custom registration function.
37 // The register_fn takes (name, type, vfs_path) and returns a handle.
38 using RegisterFn = std::function<AssetHandle(const std::string&, AssetType,
39 const std::string&)>;
40 static void ScanVfsPrefix(const std::string& prefix, RegisterFn register_fn,
41 std::vector<std::string>& scenes_to_preload);
42
43 // Apply render options to the renderer.
44 static void ApplyRenderOptions(const GameInfo& info);
45
46 // Capture current renderer settings into serialized render options.
47 static void CaptureRenderOptions(RenderOptionsSerialized& out_opts);
48
49 // Apply input settings to the input manager.
50 static void ApplyInputSettings(const GameInfo& info);
51
52 // Load the start scene into the active scene.
53 static bool LoadStartScene(const GameInfo& info);
54
55 // Convenience: mount, scan, apply settings, load start scene.
56 static bool LoadAll(const GameInfo& info,
57 const std::filesystem::path& assets_dir);
58
59 // Import a single asset: reads or creates .meta, registers in AssetManager.
60 // Creates .meta files when the VFS path has physical storage.
61 static AssetHandle ImportAsset(const std::string& name, AssetType type,
62 const std::string& vfs_path);
63
64 // Write a .meta sidecar file for a binary asset.
65 static void WriteMetaFile(const std::filesystem::path& meta_path,
66 const AssetHandle& handle, AssetType type,
67 const void* properties = nullptr);
68
69 struct MetaFileData {
70 AssetHandle handle;
71 nlohmann::json properties; // empty if none
72 };
73
74 static MetaFileData ReadMetaFile(const std::filesystem::path& meta_path);
75 static MetaFileData ReadMetaFile(const nlohmann::json& j);
76};
