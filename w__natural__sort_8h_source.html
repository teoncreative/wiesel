19// Natural sort comparator: treats embedded numbers as integers.
20// "ship_6_2" < "ship_6_10", "frame_1" < "frame_2" < "frame_10"
21inline bool NaturalLess(const std::string& a, const std::string& b) {
22 size_t i = 0;
23 size_t j = 0;
24
25 while (i < a.size() && j < b.size()) {
26 if (std::isdigit(static_cast<unsigned char>(a[i])) &&
27 std::isdigit(static_cast<unsigned char>(b[j]))) {
28 // Compare numeric segments as integers
29 // Skip leading zeros
30 size_t a_start = i;
31 size_t b_start = j;
32 while (i < a.size() && std::isdigit(static_cast<unsigned char>(a[i]))) {
33 i++;
34 }
35 while (j < b.size() && std::isdigit(static_cast<unsigned char>(b[j]))) {
36 j++;
37 }
38
39 size_t a_len = i - a_start;
40 size_t b_len = j - b_start;
41
42 // Longer number is larger (after skipping leading zeros)
43 if (a_len != b_len) {
44 return a_len < b_len;
45 }
46 // Same length: compare lexicographically (works for same-length digit strings)
47 int cmp = a.compare(a_start, a_len, b, b_start, b_len);
48 if (cmp != 0) {
49 return cmp < 0;
50 }
51 } else {
52 // Compare characters case-insensitively
53 char ca =
54 static_cast<char>(std::tolower(static_cast<unsigned char>(a[i])));
55 char cb =
56 static_cast<char>(std::tolower(static_cast<unsigned char>(b[j])));
57 if (ca != cb) {
58 return ca < cb;
59 }
60 i++;
61 j++;
62 }
63 }
64
65 return a.size() < b.size();
66}

68// Comparator struct for use with std::sort
69struct NaturalSortCompare {
70 bool operator()(const std::string& a, const std::string& b) const {
71 return NaturalLess(a, b);
72 }
73};
