126 template <typename T, typename... Args>
127 T& AddComponent(entt::entity handle, Args&&... args) {
128 if (HasComponent<T>(handle)) {
129 throw std::runtime_error("Entity already has component!");
130 }
131 auto& component = registry_.emplace<T>(handle, std::forward<Args>(args)...);
132 OnAddComponent(handle, component);

204 template <typename Entity, typename... Components, typename Func>
205 void BindSystem(SystemType type, Func func) {
206 systems_[type].push_back([this, func](float_t delta_time) {
207 for (auto handle : registry_.view<Components...>()) {
208 Entity entity{handle, this};
209 func(delta_time, entity, registry_.get<Components>(handle)...);
210 }
211 });
212 }
213
214 template <typename Entity, typename... Components, typename Func>
215 void BindSystem(SystemType type, const std::string& tag, Func func) {
216 systems_[type].push_back([this, func, tag](float_t delta_time) {
217 for (auto handle : registry_.view<Components...>()) {
218 Entity entity{handle, this};
219 if (entity.GetName() != tag) {
220 continue;
221 }
222 func(delta_time, entity, registry_.get<Components>(handle)...);
223 }
224 });
225 }
