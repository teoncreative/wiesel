114// ECS component - the "speaker" on an entity.

126 bool mute = false;
127 float spatial_blend = 0.0f; // 0 = fully 2D, 1 = fully 3D
128 float min_distance = 1.0f;
129 float max_distance = 100.0f;
130
131 // Runtime state (not serialized)
132 SoundHandle playing_handle_;
133 bool started_ = false;
134
135 // Build SoundParams from this source's settings + a world position
136 SoundParams MakeParams(const glm::vec3& position) const {
137 SoundParams p;
138 p.bus = bus;
139 p.volume = volume;
140 p.pitch = pitch;
141 p.loop = loop;
142 p.spatial_blend = spatial_blend;
143 p.position = position;
144 p.min_distance = min_distance;
145 p.max_distance = max_distance;
146 return p;
147 }
148};
