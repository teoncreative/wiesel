24struct FrustumPlanes {
25 glm::vec4 Left, Right, Bottom, Top, Near, Far;
26
27 // Test if an AABB (already in world space) is outside the frustum.
28 // Returns true if the box is completely outside any plane (should be culled).
29 bool IsBoxOutside(const glm::vec3& aabb_min,
30 const glm::vec3& aabb_max) const {
31 const glm::vec4* planes_arr = &Left;
32 for (int i = 0; i < 6; i++) {
33 glm::vec3 n(planes_arr[i]);
34 float d = planes_arr[i].w;
35 // Find the AABB corner most in the direction of the plane normal
36 glm::vec3 p_vertex(n.x >= 0 ? aabb_max.x : aabb_min.x,
37 n.y >= 0 ? aabb_max.y : aabb_min.y,
38 n.z >= 0 ? aabb_max.z : aabb_min.z);
39 if (glm::dot(n, p_vertex) + d < 0.0f) {
40 return true;
41 }
42 }
43 return false;
44 }
45};
